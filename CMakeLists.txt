cmake_minimum_required(VERSION 3.13)
project(ammio VERSION 0.6.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(AMMIO_BUILD_TESTS "Build the ammio unit tests" ON)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/std:c11 /experimental:c11atomics)
endif()

find_package(Threads REQUIRED)

include(CheckIncludeFile)
include(CheckSymbolExists)

# External dependencies

# Generated headers for vendored libraries live in the build tree, never in the submodules.
set(AMMIO_GENERATED_DIR ${CMAKE_BINARY_DIR}/generated)

add_library(cjson STATIC external/cJSON/cJSON.c)
target_include_directories(cjson PUBLIC ${CMAKE_SOURCE_DIR}/external/cJSON)

# uthash: header-only hash table
add_library(uthash INTERFACE)
target_include_directories(uthash INTERFACE ${CMAKE_SOURCE_DIR}/external/uthash/src)

# nng: messaging library

set(NNG_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory(external/nng)

# TRDP: Train Real-time Data Protocol (TCNopen), including its XML configuration reader
set(TRDP_DIR ${CMAKE_SOURCE_DIR}/external/TCNopen/trdp)
add_library(trdp STATIC
    ${TRDP_DIR}/src/common/trdp_pdcom.c
    ${TRDP_DIR}/src/common/trdp_mdcom.c
    ${TRDP_DIR}/src/common/trdp_utils.c
    ${TRDP_DIR}/src/common/trdp_stats.c
    ${TRDP_DIR}/src/common/trdp_xml.c
    ${TRDP_DIR}/src/common/tau_xml.c
    ${TRDP_DIR}/src/common/tlp_if.c
    ${TRDP_DIR}/src/common/tlc_if.c
    ${TRDP_DIR}/src/common/tlm_if.c
    ${TRDP_DIR}/src/vos/common/vos_utils.c
    ${TRDP_DIR}/src/vos/common/vos_mem.c
)
target_include_directories(trdp PUBLIC
    ${TRDP_DIR}/src/api
    ${TRDP_DIR}/src/vos/api
    ${TRDP_DIR}/src/common
)
if(WIN32)
    target_sources(trdp PRIVATE
        ${TRDP_DIR}/src/vos/windows/vos_sock.c
        ${TRDP_DIR}/src/vos/windows/vos_thread.c
        ${TRDP_DIR}/src/vos/windows/vos_shared_mem.c
    )
    target_include_directories(trdp PRIVATE ${TRDP_DIR}/src/vos/windows)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        target_compile_definitions(trdp PUBLIC WIN64 L_ENDIAN MD_SUPPORT=1)
    else()
        target_compile_definitions(trdp PUBLIC WIN32 L_ENDIAN MD_SUPPORT=1)
    endif()
    target_compile_definitions(trdp PRIVATE _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
    target_link_libraries(trdp PUBLIC ws2_32 iphlpapi)
else()
    target_sources(trdp PRIVATE
        ${TRDP_DIR}/src/vos/posix/vos_sock.c
        ${TRDP_DIR}/src/vos/posix/vos_thread.c
        ${TRDP_DIR}/src/vos/posix/vos_shared_mem.c
    )
    target_include_directories(trdp PRIVATE ${TRDP_DIR}/src/vos/posix)
    target_compile_definitions(trdp PUBLIC POSIX LINUX _GNU_SOURCE L_ENDIAN MD_SUPPORT=1)
    target_link_libraries(trdp PUBLIC Threads::Threads rt)
endif()

# libmodbus: Modbus TCP/RTU protocol
set(LIBMODBUS_VERSION_MAJOR 3)
set(LIBMODBUS_VERSION_MINOR 1)
set(LIBMODBUS_VERSION_MICRO 12)
set(LIBMODBUS_VERSION "3.1.12")
set(LIBMODBUS_DIR ${CMAKE_SOURCE_DIR}/external/libmodbus/src)
configure_file(
    ${LIBMODBUS_DIR}/modbus-version.h.in
    ${AMMIO_GENERATED_DIR}/libmodbus/modbus-version.h
    @ONLY
)
if(WIN32)
    configure_file(
        ${LIBMODBUS_DIR}/win32/config.h.win32
        ${AMMIO_GENERATED_DIR}/libmodbus_config/config.h
        @ONLY
    )
else()
    # Equivalent of libmodbus' autoconf checks for the macros its sources actually use
    set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
    check_include_file(netinet/in.h HAVE_NETINET_IN_H)
    check_include_file(netinet/ip.h HAVE_NETINET_IP_H)
    check_symbol_exists(accept4 "sys/socket.h" HAVE_ACCEPT4)
    check_symbol_exists(gai_strerror "netdb.h" HAVE_GAI_STRERROR)
    check_symbol_exists(strlcpy "string.h" HAVE_STRLCPY)
    check_symbol_exists(TIOCSRS485 "sys/ioctl.h" HAVE_DECL_TIOCSRS485)
    check_symbol_exists(TIOCM_RTS "sys/ioctl.h" HAVE_DECL_TIOCM_RTS)
    unset(CMAKE_REQUIRED_DEFINITIONS)
    set(LIBMODBUS_CONFIG "/* config.h for libmodbus - generated by CMake */\n#pragma once\n")
    set(LIBMODBUS_DEFINITIONS "")
    foreach(def HAVE_NETINET_IN_H HAVE_NETINET_IP_H HAVE_ACCEPT4 HAVE_GAI_STRERROR HAVE_STRLCPY
                HAVE_DECL_TIOCSRS485 HAVE_DECL_TIOCM_RTS)
        if(${def})
            string(APPEND LIBMODBUS_CONFIG "#define ${def} 1\n")
            # modbus-tcp.c tests some of these before it includes config.h
            list(APPEND LIBMODBUS_DEFINITIONS ${def}=1)
        endif()
    endforeach()
    file(WRITE ${AMMIO_GENERATED_DIR}/libmodbus_config/config.h.tmp "${LIBMODBUS_CONFIG}")
    configure_file(${AMMIO_GENERATED_DIR}/libmodbus_config/config.h.tmp
                   ${AMMIO_GENERATED_DIR}/libmodbus_config/config.h COPYONLY)
endif()
add_library(modbus_lib STATIC
    ${LIBMODBUS_DIR}/modbus.c
    ${LIBMODBUS_DIR}/modbus-data.c
    ${LIBMODBUS_DIR}/modbus-tcp.c
    ${LIBMODBUS_DIR}/modbus-rtu.c
)
target_include_directories(modbus_lib
    PUBLIC  ${LIBMODBUS_DIR} ${AMMIO_GENERATED_DIR}/libmodbus
    PRIVATE ${AMMIO_GENERATED_DIR}/libmodbus_config
)
target_compile_definitions(modbus_lib PUBLIC DLLBUILD)
if(WIN32)
    target_compile_definitions(modbus_lib PRIVATE FD_SETSIZE=1024)
    target_link_libraries(modbus_lib PUBLIC ws2_32)
else()
    target_compile_definitions(modbus_lib PRIVATE _GNU_SOURCE ${LIBMODBUS_DEFINITIONS})
endif()

# open62541: OPC UA protocol
set(UA_BUILD_EXAMPLES   OFF CACHE BOOL "" FORCE)
set(UA_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(UA_LOGLEVEL         600 CACHE STRING "" FORCE)
add_subdirectory(${CMAKE_SOURCE_DIR}/external/open62541)

# ==============================================================================
# ammio
# ==============================================================================

# Everything except main() lives in ammio_core so the unit tests can link against it
add_library(ammio_core STATIC
    src/util/log.c
    src/util/clock.c
    src/util/cfg.c
    src/util/codec.c
    src/core/types.c
    src/core/var_table.c
    src/core/connections.c
    src/api/api_server.c
    src/api/events.c
    src/drivers/modbus.c
    src/drivers/opcua.c
    src/drivers/trdp.c
    src/drivers/trdp_xml.c
)
target_include_directories(ammio_core PUBLIC ${CMAKE_SOURCE_DIR}/src)
target_compile_definitions(ammio_core PUBLIC AMMIO_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_definitions(ammio_core PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
    target_compile_options(ammio_core PRIVATE /W3)
else()
    target_compile_options(ammio_core PRIVATE -Wall -Wextra -Wshadow -Wno-unused-parameter)
endif()
target_link_libraries(ammio_core PUBLIC
    cjson
    uthash
    nng
    trdp
    modbus_lib
    open62541
    Threads::Threads
)

add_executable(ammio src/main.c)
target_link_libraries(ammio PRIVATE ammio_core)
